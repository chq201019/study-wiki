# 30 实战：批量处理文件

**本课目标**：写出一个能立刻用在你自己电脑上的实用工具。

## 为什么这个最实用

**批量改文件名**、**统计目录大小**、**批量转换编码**——这些手动做要几十分钟，写脚本只要几分钟，而且以后能一直用。

这是**编程最直接的价值**：把重复劳动自动化。

:::danger[批量操作前必须警告的话]
**批量重命名/删除是不可逆的。**

所以本课的脚本都带**干跑模式（dry-run）**：先只打印「将要做什么」，确认无误后再真正执行。

**养成这个习惯，能救你的数据。**
:::

## 第一步：安全地遍历目录

```python
from pathlib import Path

folder = Path(".")      # 改成你要处理的目录

# 只看当前层
for f in folder.glob("*.txt"):
    print(f.name, f.stat().st_size, "字节")
```

测试可用临时目录，别一上来就对着重要文件跑：

```python
import tempfile
from pathlib import Path

# 建一个测试目录和几个文件
tmp = Path(tempfile.mkdtemp())
for name in ["a.txt", "b.txt", "c.md"]:
    (tmp / name).write_text("内容", encoding="utf-8")

print(f"测试目录：{tmp}")
for f in tmp.glob("*.txt"):
    print(" ", f.name)
```

输出：

```text
测试目录：C:\Users\...\tmp8x2k1m
  a.txt
  b.txt
```

## 第二步：批量重命名（带干跑）

```python
from pathlib import Path


def batch_rename(folder, dry_run=True):
    """给所有 .txt 文件加上前缀

    dry_run=True 时只打印，不真的改
    """
    folder = Path(folder)
    count = 0

    for f in sorted(folder.glob("*.txt")):
        if f.name.startswith("renamed_"):
            print(f"跳过（已改过）：{f.name}")
            continue

        target = f.with_name(f"renamed_{f.name}")

        if target.exists():
            print(f"跳过（目标已存在）：{target.name}")
            continue

        print(f"{f.name}  →  {target.name}")
        if not dry_run:
            f.rename(target)
        count += 1

    mode = "干跑（未改动）" if dry_run else "已执行"
    print(f"\n[{mode}] 共处理 {count} 个文件")


if __name__ == "__main__":
    batch_rename(tmp, dry_run=True)
```

输出：

```text
a.txt  →  renamed_a.txt
b.txt  →  renamed_b.txt

[干跑（未改动）] 共处理 2 个文件
```

确认没问题后，把 `dry_run` 改成 `False` 再跑一次。

:::tip[干跑模式的三个检查点]
1. **数量对吗**：处理了 2 个，不是 200 个
2. **目标名合理吗**：没有变成奇怪的名字
3. **有没有覆盖风险**：如果目标已存在会跳过

**对着输出逐行看一遍再执行**，比对着一堆改名后的文件后悔强。
:::

## 第三步：批量统计

```python
from pathlib import Path
from collections import Counter


def analyze_folder(folder):
    """统计目录里各类文件的数量和大小"""
    folder = Path(folder)

    by_ext = Counter()      # 扩展名 → 数量
    size_by_ext = Counter() # 扩展名 → 总字节

    for f in folder.rglob("*"):
        if not f.is_file():
            continue
        ext = f.suffix.lower() or "(无扩展名)"
        by_ext[ext] += 1
        size_by_ext[ext] += f.stat().st_size

    print(f"{'类型':<12}{'数量':>6}{'总大小':>12}")
    print("-" * 32)
    for ext, count in by_ext.most_common():
        size = size_by_ext[ext]
        print(f"{ext:<12}{count:>6}{format_size(size):>12}")

    print("-" * 32)
    print(f"{'合计':<12}{sum(by_ext.values()):>6}"
          f"{format_size(sum(size_by_ext.values())):>12}")


def format_size(n):
    """把字节数转成人类可读"""
    for unit in ["B", "KB", "MB", "GB"]:
        if n < 1024:
            return f"{n:.1f} {unit}"
        n /= 1024
    return f"{n:.1f} TB"
```

输出示例：

```text
类型            数量       总大小
--------------------------------
.md              42      1.2 MB
.png             18      3.4 MB
.txt              7     12.5 KB
--------------------------------
合计             67      4.6 MB
```

:::tip[`rglob("*")` 会递归所有子目录]
- `glob("*")` 只看当前层
- `rglob("*")` 递归所有层级

统计整个项目大小时用 `rglob`。但注意**遇到巨大目录会很慢**，别对着 C 盘根目录跑。
:::

## 第四步：批量转换编码

处理从 Windows 传过来的 GBK 文件：

```python
from pathlib import Path


def convert_encoding(folder, from_enc="gbk", to_enc="utf-8", dry_run=True):
    """把目录下所有 .txt 从 GBK 转成 UTF-8"""
    folder = Path(folder)
    ok, skipped, failed = 0, 0, 0

    for f in sorted(folder.glob("*.txt")):
        try:
            text = f.read_text(encoding=from_enc)
        except UnicodeDecodeError:
            print(f"✗ 跳过（不是 {from_enc}）：{f.name}")
            skipped += 1
            continue
        except OSError as e:
            print(f"✗ 读取失败：{f.name} — {e}")
            failed += 1
            continue

        if dry_run:
            print(f"将转换：{f.name}（{len(text)} 字符）")
            ok += 1
            continue

        try:
            # 先写临时文件，成功后再替换，避免写一半损坏原文件
            tmp = f.with_suffix(".txt.tmp")
            tmp.write_text(text, encoding=to_enc)
            tmp.replace(f)
            print(f"✓ 已转换：{f.name}")
            ok += 1
        except OSError as e:
            print(f"✗ 写入失败：{f.name} — {e}")
            failed += 1

    print(f"\n成功 {ok} / 跳过 {skipped} / 失败 {failed}")
```

:::danger[为什么要写临时文件再替换]
```python
# ❌ 危险：写到一半失败，原文件就毁了
f.write_text(new_text, encoding="utf-8")

# ✅ 安全：先写临时文件，成功后才替换
tmp.write_text(new_text, encoding="utf-8")
tmp.replace(f)
```

`replace()` 是**原子操作**：要么完全成功，要么原文件不受影响。

**处理重要数据时，这个模式必须用。**
:::

## 完整工具：命令行版

把功能组合成一个真正能用的脚本：

```python
"""批量文件工具

用法：
    python filetool.py stat  <目录>
    python filetool.py rename <目录> [--apply]
    python filetool.py enc    <目录> [--apply]
"""
import sys
from collections import Counter
from pathlib import Path


def format_size(n):
    for unit in ["B", "KB", "MB", "GB"]:
        if n < 1024:
            return f"{n:.1f} {unit}"
        n /= 1024
    return f"{n:.1f} TB"


def cmd_stat(folder):
    folder = Path(folder)
    if not folder.is_dir():
        print(f"不是目录：{folder}")
        return 1

    by_ext = Counter()
    size_by_ext = Counter()
    for f in folder.rglob("*"):
        if f.is_file():
            ext = f.suffix.lower() or "(无)"
            by_ext[ext] += 1
            size_by_ext[ext] += f.stat().st_size

    print(f"{'类型':<10}{'数量':>6}{'大小':>12}")
    print("-" * 28)
    for ext, cnt in by_ext.most_common(10):
        print(f"{ext:<10}{cnt:>6}{format_size(size_by_ext[ext]):>12}")
    print("-" * 28)
    print(f"{'合计':<10}{sum(by_ext.values()):>6}"
          f"{format_size(sum(size_by_ext.values())):>12}")
    return 0


def cmd_rename(folder, apply=False):
    folder = Path(folder)
    n = 0
    for f in sorted(folder.glob("*.txt")):
        if f.name.startswith("renamed_"):
            continue
        target = f.with_name(f"renamed_{f.name}")
        if target.exists():
            print(f"跳过（已存在）：{target.name}")
            continue
        print(f"{f.name}  →  {target.name}")
        if apply:
            f.rename(target)
        n += 1
    print(f"\n{'已执行' if apply else '干跑'}：{n} 个")
    if not apply and n:
        print("确认无误后加 --apply 参数真正执行")
    return 0


def main():
    if len(sys.argv) < 3:
        print(__doc__)
        return 1

    cmd, folder = sys.argv[1], sys.argv[2]
    apply = "--apply" in sys.argv

    if cmd == "stat":
        return cmd_stat(folder)
    if cmd == "rename":
        return cmd_rename(folder, apply)
    if cmd == "enc":
        print("编码转换示例见上文，可按同样结构实现")
        return 0

    print(f"未知命令：{cmd}")
    return 1


if __name__ == "__main__":
    sys.exit(main())
```

使用：

```bash
# 先看统计
python filetool.py stat D:\code\myproject

# 干跑重命名
python filetool.py rename D:\temp\test

# 确认后执行
python filetool.py rename D:\temp\test --apply
```

## 安全清单

:::danger[批量操作前的五条铁律]
1. **先干跑**：永远先看输出，确认无误再 `--apply`
2. **先备份**：重要目录先复制一份，或者确认有 git / 版本控制
3. **先小范围测试**：拿一个临时目录跑通，再对真实目录动手
4. **注意覆盖**：检查目标文件是否已存在（`if target.exists()`）
5. **别对系统目录跑**：`C:\Windows`、`/usr` 这类地方绝对不要试
:::

## 常见错误

| 错误 | 后果 | 预防 |
| --- | --- | --- |
| 没干跑直接执行 | 文件名全乱 | 强制自己先 dry-run |
| 没检查目标存在 | **覆盖原文件** | `if target.exists(): continue` |
| 直接覆盖写文件 | 写一半损坏 | 写临时文件再 `replace` |
| `rglob` 扫了巨大目录 | 卡死 | 限定目录，加进度提示 |
| 遍历时改名 | 漏处理或重复 | 先 `sorted()` 拿到完整列表 |
| 路径含中文/空格 | 找不到文件 | 用 `Path` 而不是字符串拼接 |
